from src.monitoreo.estrategias import ETALinealStrategy


def test_eta_caso_normal():
    # 10 km a 30 km/h -> (10/30)*60 = 20.0 -> 20
    assert ETALinealStrategy().calcular_eta(10.0, 30.0) == 20


def test_eta_distancia_cero():
    assert ETALinealStrategy().calcular_eta(0.0, 30.0) == 0


def test_eta_distancia_negativa():
    assert ETALinealStrategy().calcular_eta(-5.0, 30.0) == 0


def test_eta_velocidad_muy_baja_usa_velocidad_por_defecto():
    # 1 km/h <= 5 -> v = 15; (7.5/15)*60 = 30.0 -> 30
    assert ETALinealStrategy().calcular_eta(7.5, 1.0) == 30


def test_eta_velocidad_borde_exacta_cinco_usa_velocidad_por_defecto():
    # 5.0 no es > 5.0 -> v = 15; (3/15)*60 = 12.0 -> 12
    assert ETALinealStrategy().calcular_eta(3.0, 5.0) == 12


def test_eta_velocidad_justo_sobre_el_borde_no_usa_defecto():
    # 6 km/h > 5 -> v = 6; (3/6)*60 = 30.0 -> 30
    assert ETALinealStrategy().calcular_eta(3.0, 6.0) == 30
