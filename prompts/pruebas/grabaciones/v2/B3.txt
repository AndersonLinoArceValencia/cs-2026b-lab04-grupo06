import pytest
from src.monitoreo.estrategias import ETAParadasIntermediasStrategy


def test_eta_caso_normal():
    # 10/30*60 = 20 min; paradas = 10//2 = 5 -> 10 min; total 30
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(10.0, 30.0) == 30


def test_eta_distancia_cero():
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(0.0, 30.0) == 0


def test_eta_distancia_negativa():
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(-5.0, 30.0) == 0


def test_eta_velocidad_muy_baja_usa_velocidad_por_defecto():
    # v=2 <= 5 -> v=18; 3/18*60 = 10; paradas = 3//2 = 1 -> 2; total 12
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(3.0, 2.0) == 12


def test_eta_velocidad_borde_exactamente_cinco_usa_velocidad_por_defecto():
    # v=5.0 no es > 5.0 -> v=18; 6/18*60 = 20; paradas = 3 -> 6; total 26
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(6.0, 5.0) == 26


def test_eta_velocidad_justo_sobre_el_borde_usa_velocidad_real():
    # v=5.1 > 5.0; 5.1/5.1*60 = 60; paradas = int(5.1//2) = 2 -> 4; total 64
    estrategia = ETAParadasIntermediasStrategy()
    assert estrategia.calcular_eta(5.1, 5.1) == 64
