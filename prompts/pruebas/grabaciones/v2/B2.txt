from src.monitoreo.estrategias import ETAHoraPuntaStrategy


def test_eta_caso_normal():
    # v = 30 * 0.6 = 18.0; (5 / 18) * 60 = 16.67 -> 17; 17 + 4 = 21
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(5.0, 30.0) == 21


def test_eta_distancia_cero():
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(0.0, 30.0) == 0


def test_eta_distancia_negativa():
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(-3.0, 30.0) == 0


def test_eta_velocidad_muy_baja():
    # velocidad 3.0 <= 5.0 -> v = 10.0; (2 / 10) * 60 = 12; 12 + 4 = 16
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(2.0, 3.0) == 16


def test_eta_velocidad_cero_usa_velocidad_minima():
    # velocidad 0.0 -> v = 10.0; (5 / 10) * 60 = 30; 30 + 4 = 34
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(5.0, 0.0) == 34


def test_eta_borde_velocidad_exactamente_cinco():
    # velocidad 5.0 no es > 5.0 -> v = 10.0; (1 / 10) * 60 = 6; 6 + 4 = 10
    estrategia = ETAHoraPuntaStrategy()
    assert estrategia.calcular_eta(1.0, 5.0) == 10
    # velocidad 5.1 > 5.0 -> v = 3.06; (1 / 3.06) * 60 = 19.61 -> 20; 20 + 4 = 24
    assert estrategia.calcular_eta(1.0, 5.1) == 24
